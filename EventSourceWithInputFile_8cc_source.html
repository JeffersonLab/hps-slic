68void EventSourceWithInputFile::beginRun(const G4Run*) {

70 if (EventSourceManager::instance()->isNewFilename()) {

81

83 reset();
84
85 /* Set state variables. */
86 m_fileIsNew = false;
87 m_fileIsOpen = true;
88 }
89
90 // get num events to skip from mgr
91 int nskip = EventSourceManager::instance()->getSkipEvents();
92
93 if (nskip) {
94
95#ifdef SLIC_LOG
96 log() << LOG::okay << "Skipping <" << nskip << "> events ..." << LOG::done;
97#endif
98
99 // skip events according to current mgr val
100 int actualskipped = skipEvents(nskip);
101
102#ifdef SLIC_LOG
103 log() << LOG::okay << "Skipped <" << actualskipped << "> events." << LOG::done;
104
105 if (actualskipped < nskip) {
106 log() << LOG::warning << "Ran out of records to skip on event file!" << LOG::done;
107 }
108#endif
109
110 // reset skip events
111 EventSourceManager::instance()->setSkipEvents(0);
