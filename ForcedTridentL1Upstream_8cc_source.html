39void ForcedTridentL1Upstream::beginRun(const G4Run *) {
40 if (initialized_) {
41 return;
42 }
43
44 // G4ePairProduction is not part of QGSP_BERT's default electron physics.
45 // This plugin owns a private instance purely as a sampling engine --
46 // CrossSectionPerVolume()/SelectModelForMaterial()/SampleSecondaries()
47 // are all called manually from stepping() below, exactly as the existing
48 // ForcedPairConversionL1Upstream plugin calls the real "conv" process.
49 // Deliberately NOT attached via AddDiscreteProcess(): doing so makes
50 // Geant4's own per-step dispatch (G4SteppingManager->
51 // PostStepGetPhysicalInteractionLength->ComputeLambdaForScaledEnergy)
52 // invoke this process on every electron step, which segfaults because
53 // the lambda-table infrastructure that path depends on is never built
54 // (only PreparePhysicsTable(), not BuildPhysicsTable(), is called here,
55 // deliberately, to avoid touching shared dE/dx tables -- see below). No
56 // shared physics-list code is touched either way.
57 pairProd_ = new G4ePairProduction();
58
59 // PreparePhysicsTable() (not BuildPhysicsTable(), which would rebuild
60 // dE/dx and range tables for every registered electron energy-loss
61 // process) is sufficient to construct and initialise the underlying
62 // G4MuPairProductionModel so that CrossSectionPerVolume(),
63 // SelectModelForMaterial() and SampleSecondaries() all work correctly
64 // below -- CrossSectionPerVolume() is intended to fall back to a direct
65 // model evaluation when no lambda table has been built. However,
66 // PreparePhysicsTable() itself leaves theLambdaTable pointing at a
67 // non-null but still-empty G4PhysicsTable (allocated by
68 // G4PhysicsTableHelper::PreparePhysicsTable(), never populated since
69 // BuildPhysicsTable() is deliberately never called), which otherwise
70 // makes CrossSectionPerVolume() wrongly take the lambda-table lookup
71 // branch and segfault dereferencing an empty table. Null it out so the
72 // intended direct-model-evaluation fallback path is actually taken.
73 pairProd_->PreparePhysicsTable(*G4Electron::Electron());
74 pairProd_->SetLambdaTable(nullptr);
75
76 initialized_ = true;
77}

79void ForcedTridentL1Upstream::stepping(const G4Step *step) {
80
81 if (hasInteracted_) {
82 // Primary beam electron has already been forced to interact this
83 // event; everything else is left to propagate under normal physics.
84 return;
85 }
86
87 G4Track *track = step->GetTrack();
88
89 // Only the primary (generator-level) beam electron is ever a candidate
90 // for forcing.
91 if (track->GetDefinition() != G4Electron::Electron() || track->GetParentID() != 0) {
92 return;
93 }
94
95 G4VPhysicalVolume *volume = track->GetVolume();
96 if (volume == nullptr) {
97 return;
98 }
99 const G4String &volumeName = volume->GetName();
100 bool inTargetSensor = false;
101 for (const std::string &target : kTargetVolumes) {
102 if (volumeName.find(target) != std::string::npos) {
103 inTargetSensor = true;
104 break;
105 }
106 }
107 if (!inTargetSensor) {
108 // Not yet in (or already past) the forcing sensor.
109 return;
110 }
111
112 const G4double kinEnergy = track->GetKineticEnergy();
113 const G4double logKinEnergy = track->GetDynamicParticle()->GetLogKineticEnergy();
114 const G4MaterialCutsCouple *couple = track->GetMaterialCutsCouple();
115
116 // Local interaction probability for this slab: 1 - exp(-t/lambda), with
117 // 1/lambda given by the real cross section per unit volume. This is the
118 // weight the event is worth once we force the trident to happen here.
119 // The 3-arg overload is called explicitly (mirroring
120 // G4VEnergyLossProcess::MeanFreePath()'s own usage) because the 2-arg
121 // overload, while declared in the installed Geant4 10.6.p01 header, has
122 // no corresponding definition/export in the installed shared library.
123 const G4double sigma = pairProd_->CrossSectionPerVolume(kinEnergy, couple, logKinEnergy);
124 const G4double stepLength = step->GetStepLength();
125 const G4double weight = 1.0 - std::exp(-sigma * stepLength);
126
127 // Sample the e+/e- pair kinematics from the same model Geant4 would have
128 // used. SampleSecondaries() only appends the two pair secondaries to the
129 // output vector -- the degraded primary's kinematics are written by the
130 // model to the process's own internal (protected) particle-change object,
131 // which isn't reachable from here, so the scattered primary electron's
132 // final 4-momentum is instead reconstructed below from 4-momentum
133 // conservation using the two sampled secondaries.
134 size_t idx = (size_t)couple->GetIndex();
135 G4VEmModel *model = pairProd_->SelectModelForMaterial(kinEnergy, idx);
136
137 std::vector<G4DynamicParticle *> secondaryParticles;
138 model->SampleSecondaries(&secondaryParticles, couple, track->GetDynamicParticle());
139
140 if (secondaryParticles.size() != 2) {
141 // Nothing physical was sampled; leave the primary untouched.
142 return;
143 }
144
145 const G4LorentzVector primaryIn4Momentum = track->GetDynamicParticle()->Get4Momentum();
146 G4LorentzVector secondaries4Momentum(0., 0., 0., 0.);
147 for (G4DynamicParticle *dp : secondaryParticles) {
148 secondaries4Momentum += dp->Get4Momentum();
149 }
150 // Nuclear recoil carries away the (small) remaining momentum imbalance;
151 // neglecting it here and assigning everything else to the scattered
152 // primary electron is the same order of approximation already accepted
153 // for the forced pair-conversion plugins' vertex placement.
154 const G4LorentzVector scatteredElectron4Momentum = primaryIn4Momentum - secondaries4Momentum;
155
156 G4DynamicParticle *scatteredElectron =
157 new G4DynamicParticle(G4Electron::Electron(), scatteredElectron4Momentum);
158
159 // Place the vertex at a random depth within the slab rather than always
160 // at the exit face, so the resulting tracks can deposit charge in this
161 // sensor too -- same approximation already accepted for the forced
162 // pair-conversion plugins' "randpos" variant.
163 const G4ThreeVector &entryPosition = step->GetPreStepPoint()->GetPosition();
164 const G4ThreeVector &exitPosition = step->GetPostStepPoint()->GetPosition();
165 const G4double entryTime = step->GetPreStepPoint()->GetGlobalTime();
166 const G4double exitTime = step->GetPostStepPoint()->GetGlobalTime();
167
168 const G4double u = G4UniformRand();
169 const G4ThreeVector position = entryPosition + u * (exitPosition - entryPosition);
170 const G4double globalTime = entryTime + u * (exitTime - entryTime);
171
172 // Push onto the current step's own secondary bucket -- the native Geant4
173 // hook for adding secondaries during a stepping action (see
174 // ForcedPairConversionL1Upstream.cc for the detailed rationale: this is
175 // what gets each forced track a real, non-zero track ID via
176 // G4EventManager::StackTracks()).
177 G4TrackVector *secondaryBucket = const_cast<G4Step *>(step)->GetfSecondary();
178
179 G4Track *scatteredElectronTrack = new G4Track(scatteredElectron, globalTime, position);
180 scatteredElectronTrack->SetParentID(track->GetTrackID());
181 secondaryBucket->push_back(scatteredElectronTrack);
182
183 for (G4DynamicParticle *dp : secondaryParticles) {
184 G4Track *secondaryTrack = new G4Track(dp, globalTime, position);
185 secondaryTrack->SetParentID(track->GetTrackID());
186 secondaryBucket->push_back(secondaryTrack);
187 }
188
189 track->SetTrackStatus(fStopAndKill);
190
191 hasInteracted_ = true;
192 ++forcedCount_;
193
194 // Stash the weight on the MCParticle collection under the key that
195 // LcioManager::createLCEvent() reads at end-of-event. Unlike the
196 // stdhep-driven ForcedPairConversion* plugins, this plugin's GPS-driven
197 // gun never calls MCParticleManager::generateEvent() (GPS bypasses
198 // MCParticleManager entirely), so the collection doesn't exist yet at
199 // this point in the event and must be created here. This is safe because
200 // TrackManager::saveTrackSummaries() now determines its geant4Generator
201 // flag from the actual event source type (EventSourceManager), not from
202 // whether the collection already exists -- see that file.
203 MCParticleManager::instance()->createMCParticleCollection();
204 LCCollectionVec *mcpColl = MCParticleManager::instance()->getMCParticleCollection();
205 mcpColl->parameters().setValue("_weight", (float)weight);
206
207 if (verbose_ > 1) {
208 G4cout << "[ ForcedTridentL1Upstream ]: forced trident in " << volumeName << " with weight "
209 << weight << G4endl;
210 }
211}
