40std::string LcioFileNamer::getFieldValue(std::string field) {
41 std::string value;
42
43 static std::string sep("-");
44
45 if (field == "application") {
46 value = PackageInfo::getShortName() + "-v" + PackageInfo::getVersion(sep);
47 } else if (field == "geometry") {
48 value = LCDDProcessor::instance()->getDetectorName();
49 } else if (field == "date") {
50 value = TimeUtil::getDate();
51 } else if (field == "event" || field == "evt") {
52 value = LcioFileNamer::makeEventName();
53 } else if (field == "eventNumber") {
54 value = LcioFileNamer::makeEventNumberString();
55 } else if (field == "run") {
56 value = LcioFileNamer::makeRunNumberString();
57 } else if (field == "binary") {
58 value = SlicApplication::instance()->getBinaryBasename();
59 } else if (field == "physics") {
60 value = PhysicsListManager::instance()->getCurrentListName();
61 } else {
62#ifdef SLIC_LOG
63 log() << LOG::error << "Ignoring unknown autoname field <" << field << ">." << LOG::done;
64#endif
65 value = "";
66 }
67 return value;
68}

70std::string LcioFileNamer::makeFileName(std::vector<std::string> fieldList) {
71 std::string filename = "";
72 if (fieldList.size() != 0) {
73 for (std::vector<std::string>::const_iterator it = fieldList.begin(); it != fieldList.end(); it++) {
74 std::string value = getFieldValue(*it);
75 if (value != "") {
76 filename = filename + value + m_sep;
77 }
78 }
79 if (filename != "") {
80 filename.erase(filename.end() - 1, filename.end());
81 }
82 } else {
83 filename = makeDefaultFileName();
84 }
85 return filename;

92std::string LcioFileNamer::makeEventName() {
93 std::string evt;
94
95 EventSourceManager::ESourceType est = EventSourceManager::instance()->getCurrentSourceType();
96
97 if (EventSourceManager::instance()->isFileSource()) {
98 evt = makeFileBasedName();
99 } else if (est == EventSourceManager::eGPS) {
100 evt = makeGPSName();
101 } else if (est == EventSourceManager::eParticleGun) {
102 evt = makeGunName();
103 } else {
104 evt = "events";
105 }
106 return evt;
