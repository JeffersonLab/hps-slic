31// Downstream (second) sensor per half of Layer 2 -- same axial/stereo-vs-
32// layer-parity convention as Layer 1 (top: stereo is downstream, bottom:
33// axial is downstream). Matching only the sensor volumes (not the
34// halfmodule wrapper) means the forced conversion always lands in actual
35// silicon.
36const std::vector<std::string> ForcedPairConversionL2Downstream::kTargetVolumes = {
37 "module_L2t_halfmodule_stereo_sensor",
38 "module_L2b_halfmodule_axial_sensor",
39};

41// All six upstream sensor volumes: all four Layer 1 sensors (both the
42// upstream and downstream sensor of each half) plus both Layer 2 upstream

84void ForcedPairConversionL2Downstream::stepping(const G4Step *step) {
85
86 G4Track *track = step->GetTrack();
87
88 // Only the primary (generator-level) photon is ever a candidate for
89 // forcing, or relevant to the upstream-volume deactivation toggle below.
90 // Any photon with a parent (e.g. from eBrem on a forced e+/e-) is skipped
91 // here, so a forced conversion can never cascade downstream.
92 if (track->GetDefinition() != G4Gamma::Gamma() || track->GetParentID() != 0) {
93 return;
94 }
95
96 // Suppress natural conversion while the primary photon transits any of
97 // the six upstream sensor volumes, and restore it the moment it leaves
98 // the last one, so every event's photon survives all of Layer 1 and the
99 // Layer 2 upstream sensor intact to reach the Layer 2 downstream target
100 // instead of some fraction converting naturally along the way. Checked
101 // ahead of the hasConverted_ guard below since this must run on every
102 // step of the primary photon's track, not just while it's still a
103 // forcing candidate.
104 //
105 // Deliberately keyed off GetNextVolume() (the volume the upcoming step
106 // will be in), not GetVolume() (the volume the just-finished step was
107 // in): stepping() runs after a step's physics already executed, so by the
108 // time GetVolume() reports "inside an upstream sensor", any natural
109 // conversion during that very step has already happened -- one step too
110 // late to suppress it. GetNextVolume() lets the toggle take effect before
111 // the photon's first step inside the sensor.
112 G4VPhysicalVolume *volume = track->GetVolume();
113 G4VPhysicalVolume *nextVolume = track->GetNextVolume();
114 const bool enteringOrInUpstream = nextVolume != nullptr && matchesAnyVolume(nextVolume->GetName(), kUpstreamVolumes);
115 if (enteringOrInUpstream != convDeactivated_) {
116 G4VEmProcess *conv = findConversionProcess(step);
117 if (conv != nullptr) {
118 track->GetDefinition()->GetProcessManager()->SetProcessActivation(conv, !enteringOrInUpstream);
119 convDeactivated_ = enteringOrInUpstream;
120 }
121 }
122
123 if (hasConverted_) {
124 // Primary photon has already converted (naturally or forced) this
125 // event; everything else (including any secondary photons produced by
126 // the resulting e+/e-) is left to propagate under normal physics.
127 return;
128 }
129
130 // If the photon really converted on its own in this step, we're done:
131 // nothing to force, and the event is unweighted (weight stays at 1).
132 const std::vector<const G4Track *> *secondaries = step->GetSecondaryInCurrentStep();
133 if (secondaries->size() > 0 &&
134 (*secondaries)[0]->GetCreatorProcess() != nullptr &&
135 (*secondaries)[0]->GetCreatorProcess()->GetProcessName() == "conv") {
136 hasConverted_ = true;
137 return;
138 }
139
140 if (volume == nullptr) {
141 return;
142 }
143 const G4String &volumeName = volume->GetName();
144 if (!matchesAnyVolume(volumeName, kTargetVolumes)) {
145 // Not yet in (or already past) the forcing sensor.
146 return;
147 }
148
149 // Only force at the far boundary of the layer, i.e. once we know the
150 // photon transited the whole slab without converting naturally.
151 if (step->GetPostStepPoint()->GetStepStatus() != fGeomBoundary) {
152 return;
153 }
154
155 G4VEmProcess *conv = findConversionProcess(step);
156 if (conv == nullptr) {
157 return;
158 }
159
160 const G4double kinEnergy = track->GetKineticEnergy();
161 const G4MaterialCutsCouple *couple = track->GetMaterialCutsCouple();
162
163 // Local conversion probability for this slab: 1 - exp(-t/lambda), with
164 // 1/lambda given by the real cross section per unit volume. This is the
165 // weight the event is worth once we force the conversion to happen here.
166 const G4double sigma = conv->CrossSectionPerVolume(kinEnergy, couple);
167 const G4double stepLength = step->GetStepLength();
168 const G4double weight = 1.0 - std::exp(-sigma * stepLength);
169
170 // Sample the actual e+/e- kinematics from the same model Geant4 would
171 // have used, then hand the secondaries to the track stack and kill the
172 // photon -- the net effect is a conversion at this point, but driven by
173 // the real model rather than hand-rolled kinematics.
174 size_t idx = (size_t)couple->GetIndex();
175 G4VEmModel *model = conv->SelectModelForMaterial(kinEnergy, idx);
176
177 std::vector<G4DynamicParticle *> secondaryParticles;
178 model->SampleSecondaries(&secondaryParticles, couple, track->GetDynamicParticle());
179
180 // Place the conversion at a random depth within the slab rather than
181 // always at the exit face, so the resulting e+/e- can deposit charge in
182 // this sensor too. The conversion probability per slab is small, so a
183 // uniform draw along the chord is an adequate stand-in for the (more
184 // correct but barely different) exponential-attenuation distribution.
185 const G4ThreeVector &entryPosition = step->GetPreStepPoint()->GetPosition();
186 const G4ThreeVector &exitPosition = step->GetPostStepPoint()->GetPosition();
187 const G4double entryTime = step->GetPreStepPoint()->GetGlobalTime();
188 const G4double exitTime = step->GetPostStepPoint()->GetGlobalTime();
189
190 const G4double u = G4UniformRand();
191 const G4ThreeVector position = entryPosition + u * (exitPosition - entryPosition);
192 const G4double globalTime = entryTime + u * (exitTime - entryTime);
193
194 // Push onto the current step's own secondary bucket (the same G4TrackVector
195 // that G4VEmModel::SampleSecondaries() and friends use internally, exposed
196 // via G4Step::GetfSecondary()/NewSecondaryVector()). This is the native
197 // Geant4 hook for adding secondaries during a stepping action: after this
198 // UserSteppingAction call returns, G4SteppingManager/G4TrackingManager
199 // harvest this same vector (via GimmeSecondaries()) and hand it to
200 // G4EventManager::StackTracks(), which is what assigns each track a real
201 // (non-zero) track ID via its internal counter. Calling StackTracks()
202 // directly isn't an option -- it's a private method -- and pushing onto
203 // the G4StackManager directly (the original approach) bypasses ID
204 // assignment entirely, leaving every forced secondary with the default
205 // trackID of 0. That in turn breaks LCIO's SimTrackerHit->MCParticle
206 // lookup once more than one forced conversion occurs in a run.
207 G4TrackVector *secondaryBucket = const_cast<G4Step *>(step)->GetfSecondary();
208 for (G4DynamicParticle *dp : secondaryParticles) {
209 G4Track *secondaryTrack = new G4Track(dp, globalTime, position);
210 secondaryTrack->SetParentID(track->GetTrackID());
211 secondaryBucket->push_back(secondaryTrack);
212 }
213
214 track->SetTrackStatus(fStopAndKill);
215
216 hasConverted_ = true;
217 ++forcedCount_;
218
219 // Stash the weight on the MCParticle collection under the key that
220 // LcioManager::createLCEvent() reads at end-of-event
221 // (LcioManager.cc:331), since by the time any plugin's endEvent() hook
222 // runs the LCEvent has already been created and its weight already read.
223 LCCollectionVec *mcpColl = MCParticleManager::instance()->getMCParticleCollection();
224 if (mcpColl != nullptr) {
225 mcpColl->parameters().setValue("_weight", (float)weight);
226 }
227
228 if (verbose_ > 1) {
229 G4cout << "[ ForcedPairConversionL2Downstream ]: forced conversion in " << volumeName
230 << " with weight " << weight << G4endl;
231 }
232}

234void ForcedPairConversionL2Downstream::endEvent(const G4Event *) {
235 // Defensive restore: normally the stepping action re-activates "conv" the
236 // moment the primary photon leaves the last upstream volume, but if that
237 // track is ever killed by some other process while still inside one
238 // (e.g. photoelectric absorption), no further step would run the restore
239 // -- leaving conv deactivated for every other gamma, potentially leaking
240 // into the next event.
241 if (convDeactivated_ && convProcess_ != nullptr) {
242 G4Gamma::Gamma()->GetProcessManager()->SetProcessActivation(convProcess_, true);
243 }
244 convDeactivated_ = false;
245 hasConverted_ = false;
246}
