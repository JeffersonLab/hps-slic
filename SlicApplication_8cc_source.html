81void SlicApplication::initialize() {
82 if (!m_isInitialized) {
83 // Print the SLIC splash screen.
84 printSplashScreen();
85
86 // Initialize the application messenger.
87 m_appMessenger = new SlicApplicationMessenger(this);
88
89 // Create a new RunManager.
90 m_runManager = new RunManager();
91
92 PhysicsListManager::instance();
93
94 // Initialize the LcioManager.
95 LcioManager::instance();
96
97 // Initialize the magnetic field messenger.
98 FieldManager::instance();
99
100 // Setup the LCDD subsystem.
101 initializeLCDD();
102
103 // Initialize the tracker manager.
104 TrackManager::instance();
105
106 // Initialize the MCParticle manager.
107 MCParticleManager::instance();
108
109 // Initialize visualization.
110 initializeVis();
111
112 // Set state variable.
113 m_isInitialized = true;
114 } else {
115 log().warning("Call to SlicApplication::initialize() was ignored. SLIC is already initialized!");
116 }
117}

174void SlicApplication::run() {
175 // Initialize if not already.
176 if (!m_isInitialized) {
177 initialize();
178 }
179
180 // Retrieve the list of Geant4 commands created by CommandLineProcessor.
181 CommandQueue* q = CommandLineProcessor::instance()->getCommandQueue();
182
183 // Print, execute, and clear the CommandQueue.
184#ifdef SLIC_LOG
185 log().verbose("Printing the Geant4 command queue ...");
186 q->printOut();
187#endif
188 q->execute();
189 q->clear();
190
191 // Start the UI session if in interactive mode.
192 if (getMode() == eInteractive) {
193 log().verbose("Starting interactive session ...");
194 m_ui->SessionStart();
195 delete m_ui;
196 }
197}
