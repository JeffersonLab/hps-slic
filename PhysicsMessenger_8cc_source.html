24void PhysicsMessenger::SetNewValue(G4UIcommand* cmd, G4String newVals) {
25 std::istringstream is((const char*) newVals);
26
27 std::string s;
28 is >> s;
29
30 PhysicsListManager* mgr = PhysicsListManager::instance();
31
32 // set physics list
33 if (cmd == m_selectCmd) {
34 mgr->setCurrentListName(s);
35 }
36 // print available lists
37 else if (cmd == m_printListsCmd) {
38 PhysicsListManager::instance()->printAvailablePhysicsLists();
39 }
40 // Print current list and whether initialized.
41 else if (cmd == m_printCurrentCmd) {
42 mgr->log() << LOG::okay << "Current physics list <" << mgr->getCurrentListName() << ">." << LOG::done;
43 mgr->log() << LOG::okay << "Physics has been initialized <" << mgr->isInitialized() << ">." << LOG::done;
44 }
45 // Enable optical physics processes.
46 else if (cmd == m_enableOpticalCmd) {
47 bool enableOptical = true;
48 if (newVals != G4String(""))
49 enableOptical = G4UIcmdWithABool::GetNewBoolValue(newVals);
50 PhysicsListManager::instance()->enableOptical(enableOptical);
51 }
52 // Set location of particle.tbl for HepPDT.
53 else if (cmd == m_pdgCmd) {
54 HepPDTManager::instance()->setParticleDataFile(s);
55 } else {
56 mgr->log() << LOG::error << "Unknown command for PhysicsMessenger." << LOG::done;
57 }
