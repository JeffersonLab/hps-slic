189 LogStream& operator <<(const LOG::EManip& x) {
190 if (isActive()) {
191 // End of message.
192 if (x == LOG::done) {
193 getOutputStream() << std::endl;
194 // Print full header info.
195 } else if (x == LOG::head) {
196 getOutputStream() << TimeUtil::getAscTime() << " :: " << levelNames[m_currentMessageLevel] << " :: " << getName() << " :: ";
197 // Print current time.
198 } else if (x == LOG::time) {
199 getOutputStream() << TimeUtil::getAscTime();
200 // Print current message level.
201 } else if (x == LOG::level) {
202 getOutputStream() << levelNames[m_currentMessageLevel];
203 // Print logger name.
204 } else if (x == LOG::name) {
205 getOutputStream() << getName();
206 } else if (x == LOG::sep) {
207 getOutputStream() << " :: ";
208 // Print end of line terminator.
209 } else if (x == LOG::endl) {
210 getOutputStream() << std::endl;
211 }
212 }
213 return *this;
214 }
