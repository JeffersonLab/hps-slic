81void LcioManager::openLcioFile() {
82
83 // Get full output path with extension to check its existence.
84 string fullFilename = getFullOutputPath(true);
85
86 // Default to writing a new file.
87 int writeMode = LCIO::WRITE_NEW;
88
89 // File exists?
90 if (FileUtil::fileExists(fullFilename)) {
91
92 // Failure mode is on, so kill the current run.
93 if (m_fileExistsAction == eFail) {
94 log() << LOG::fatal << LOG::name << LOG::sep << "LCIO file already exists: " << fullFilename << LOG::done;
95 RunManager::instance()->abortRun(SlicApplication::OUTPUT_FILE_EXISTS);
96 // Deletion mode is on, so try to remove the current file.
97 } else if (m_fileExistsAction == eDelete) {
98 if (FileUtil::removeFile(fullFilename) != 0) {
99 // Fatal error. File could not be removed.
100 log() << LOG::fatal << LOG::head << "Unable to delete old LCIO file: " << fullFilename << LOG::done;
101 RunManager::instance()->abortRun(SlicApplication::FAILED_DELETE_LCIO_FILE);
102 } else {
103 log().okay("Deleted old LCIO file: " + fullFilename);
104 }
105 }
106 // Append mode is on.
107 else if (m_fileExistsAction == eAppend) {
108 log().okay("Appending to existing LCIO file: " + fullFilename);
109 writeMode = LCIO::WRITE_APPEND;
110 }
111 }
112 // Create a new file if doesn't exist.
113 else {
114 log().okay("Creating new LCIO file: " + fullFilename);
115 }
116
117 // Open the file using the writer with the full path.
118 if (m_writer) {
119 if (!RunManager::instance()->isRunAborted()) {
120 m_writer->open(getFullOutputPath(false), writeMode);
121 m_writerIsOpen = true;
122 }
123 } else {
124 //G4Exception("", "", FatalException, "LCWriter is null.");
125 log() << LOG::fatal << LOG::head << "LCWriter is null." << LOG::done;
126 RunManager::instance()->abortRun();
127 }
128}

130LcioManager::EFileExistsAction LcioManager::getFileExistsActionFromString(const string &feaStr) {
131 string s = StringUtil::toLower(feaStr);
132 EFileExistsAction fea = eInvalid;
133 if (s == "fail") {
134 fea = eFail;
135 } else if (s == "overwrite" || s == "delete") {
136 fea = eDelete;
137 } else if (s == "append") {
138 fea = eAppend;
139 }
140
141 return fea;
142}

167string LcioManager::getFullOutputPath(bool withExtension) {
168 string fullPath;
169
170 if (m_path.length() > 0) {
171 // Append the output directory if one exists.
172 fullPath = m_path;
173 } else {
174 // Default to the current directory if none specified.
175 fullPath = ".";
176 }
177
178 // Append file name.
179 fullPath += "/" + m_filename;
180
181 // Append extension.
182 if (withExtension) {
183 fullPath += ".slcio";
184 }
185
186 return fullPath;

189void LcioManager::beginRun(const G4Run* aRun) {
190
191 // Set the G4Run counter.
192#if ( G4VERSION_NUMBER >= 1000 )
193 G4RunManager::GetRunManager()->GetNonConstCurrentRun()->SetRunID(m_runNumber);
194#else
195 // Alternate code for backward compatibility.
196 const_cast<G4Run *>(G4RunManager::GetRunManager()->GetCurrentRun())->SetRunID(m_runNumber);
197#endif
198
199 // Automatically create LCIO output file name if option was selected.
200 if (m_usingAutoname) {
201 makeAutoname();
202 }
203
204 // Open the LCIO output file for writing.
205 openLcioFile();
206
207 // Run aborted, because LCIO output file already exists.
208 if (!RunManager::instance()->isRunAborted()) {
209
210 // create new LCRunHeader
211 createRunHeader(aRun);
212
213 // write the run header
214 m_writer->writeRunHeader(m_runHdr);
215
216 // Incr run number.
217 ++m_runNumber;
218 }
219}

221void LcioManager::endRun(const G4Run*) {
222
223 // Delete the LCIO run header object.
224 deleteRunHeader();
225
226 // Set append mode for subsequent writes if interactive mode and the run was not aborted.
227 if (!RunManager::instance()->isRunAborted()) {
228 // If run was aborted, writer was never opened.
229 if (m_writer != 0) {
230 m_writer->close();
231 m_writerIsOpen = false;
232 }
233 // Close the writer to flush it (could be reopened).
234 if (SlicApplication::instance()->getMode() == SlicApplication::eInteractive) {
235 m_fileExistsAction = eAppend;
236 }
237 }

317LCEventImpl* LcioManager::createLCEvent(const G4Event *anEvent) {
318
319 /* Create the LCEvent and set basic parameters. */
320 LCEventImpl* lcevt = new LCEventImpl();
321 lcevt->setEventNumber(anEvent->GetEventID());
322 lcevt->setRunNumber(m_runHdr->getRunNumber());
323 lcevt->setDetectorName(m_runHdr->getDetectorName());
324
325 /* Get the current MCParticle collection. */
326 LCCollection* mcpColl = MCParticleManager::instance()->getMCParticleCollection();
327
328 if (mcpColl != 0) {
329
330 /* Set the event weight. */
331 float eventWeight = mcpColl->getParameters().getFloatVal("_weight");
332 if (eventWeight != 0.0) {
333 lcevt->setWeight(eventWeight);
334 }
335
336 /* Set the idrup. */
337 int idrup = mcpColl->getParameters().getIntVal("_idrup");
338 //G4cout << "idrup = " << idrup << G4endl;
339 if (idrup != 0) {
340 lcevt->parameters().setValue("_idrup", idrup);
341 }
342 }
343
344 // Write SLIC version into event header.
345 lcevt->parameters().setValue("SLIC_VERSION", PackageInfo::getVersion());
346
347 /* Set the current LCEvent. */
348 setCurrentLCEvent(lcevt);
349
350 return lcevt;
351}
