35void GeneratorMessenger::SetNewValue(G4UIcommand* cmd, G4String newVals) {
36 std::istringstream is((const char*) newVals);
37
38 std::string s;
39 is >> s;
40
41 EventSourceManager* mgr = EventSourceManager::instance();
42
43 // select
44 if (cmd == m_selectCmd) {
45 mgr->setupEventSource(s);
46 }
47 // reset
48 else if (cmd == m_resetCmd) {
49 mgr->resetCurrentEventSource();
50 }
51 // skipEvents
52 else if (cmd == m_skipEventsCmd) {
53 mgr->setSkipEvents(StringUtil::toInt(s));
54 }
55 // filename
56 else if (cmd == m_filenameCmd) {
57 mgr->setFilename(s);
58 }
59 // dumpCurrentEvent
60 else if (cmd == m_enablePrintEventCmd) {
61 G4bool enablePrintEvent = StringUtil::toBool(s);
62 mgr->enablePrintEvent(enablePrintEvent);
63 }
64 // printNumEventsGenerated
65 else if (cmd == m_printNumEventsGeneratedCmd) {
66 mgr->printNumEventsGenerated();
67 }
68 // set random seed
69 else if (cmd == m_randomSeedCmd) {
70 G4int seed = 0;
71
72 // seed from user arg
73 if (s != std::string("")) {
74 seed = StringUtil::toInt(s);
75 }
76 // seed from time
77 // @todo Get MS instead of just seconds, which is not very good quality for a seed.
78 else {
79 seed = ((unsigned) time(NULL));
80 }
81
82 // seed it
83 HepRandom::setTheSeed(seed);
84
85 mgr->log() << LOG::okay << "set random seed: " << seed << LOG::done;
86 }
87 // Lorentz transformation angle
88 else if (cmd == m_setLorentzTransformationAngleCmd) {
89 EventSourceManager::instance()->setLorentzTransformationAngle(G4UIcmdWithADoubleAndUnit::GetNewDoubleValue(newVals));
90 } else if (cmd == m_setZSmearingParameterCmd) {
91 EventSourceManager::instance()->setZSmearing(G4UIcmdWithADouble::GetNewDoubleValue(newVals));
92 } else {
93 G4Exception("", "", JustWarning, "Unknown command.");
94 }
95}
