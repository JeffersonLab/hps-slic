62EventSourceManager::ESourceType EventSourceManager::getSourceTypeFromFileExtension(const std::string& fext) {
63 ESourceType est = eUnknown;
64
65 /* LCIO file extension */
66 if (fext == "slcio") {
67 est = eLCIO;
68 }
69 /* StdHep files, which may have .xdr file ext */
70 else if (fext == "stdhep" || fext == "xdr") {
71 est = eStdHep;
72 /* LHE files */
73 } else if (fext == "lhe") {
74 est = eLHE;
75 } else {
76 log() << LOG::error << "WARNING: File <" << m_filename << "> does not have a known file extension." << LOG::done;
77 }
78 return est;
79}

184EventSource* EventSourceManager::createEventSource(ESourceType st) {
185 EventSource* src = 0;
186 if (st == eLCIO) {
187 src = new LcioEventSource(getFilename());
188 } else if (st == eStdHep) {
189 src = new StdHepEventSource(getFilename());
190 } else if (st == eGPS) {
191 src = new GPSEventSource();
192 } else if (st == eParticleGun) {
193 return m_particleGunSource;
194 } else if (st == eLHE) {
195 return new slic::LHEEventSource(getFilename());
196 } else if (st == eUnknown) {
197 log() << LOG::error << "The event source type <" << getSourceNameFromType(st) << " is flagged as unknown!" << LOG::done;
198 } else {
199 log() << LOG::error << "Invalid ESourceType." << LOG::done;
200 }
201
202 return src;
203}

205void EventSourceManager::setupEventSource(ESourceType st) {
206 // Is it a known source type?
207 if (st != eUnknown) {
208 // It is a new type of source?
209 if (isNewSource(st)) {
210 // Set new source type.
211 m_sourceType = st;
212
213 // Delete current source.
214 deleteCurrentEventSource();
215
216 // Create new source.
217 m_currentEventSource = createEventSource(m_sourceType);
218
219#ifdef SLIC_LOG
220 log().okay("Created event generator <" + getCurrentSourceName() + ">");
221#endif
222 } else {
223 log().warning("Type of new event source is same as old; keeping old source.");
224 }
225 } else {
226 G4Exception("", "", JustWarning, "Event Source type is unknown.");
227 }
228}

259void EventSourceManager::beginRun(const G4Run* aRun) {
260
261#ifdef SLIC_LOG
262 log() << LOG::debug << LOG::name << "beginRun" << LOG::done;
263#endif
264
265 // Setup a new event source if neccessary.
266 setupEventSource();
267
268 // Call the beginRun() function of the current event source.
269 if (m_currentEventSource != 0) {
270 m_currentEventSource->beginRun(aRun);
271
272 // Filename is now old so source won't reset.
273 m_newFilename = false;
274
275 // Reset event counter.
276 m_ngen = 0;
277 } else {
278 log() << LOG::fatal << "No event generator was setup." << LOG::done;
279 RunManager::instance()->abortRun(SlicApplication::GENERATOR_NOT_SETUP);
280 }
281}
